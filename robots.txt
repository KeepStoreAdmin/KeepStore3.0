# KeepStore 3.0 - Robots policy
# Obiettivo: consentire il crawling delle pagine pubbliche (catalogo/contenuti) e bloccare aree non indicizzabili
# (account, carrello/checkout, azioni) per evitare duplicati, leak di URL sensibili e crawl trap.
User-agent: *

# Block action URLs / faceted traps
Disallow: /*rimuovi=
Disallow: /*?rimuovi=
Disallow: /*&rimuovi=

# NOTE: non blocchiamo st= (settore) perché fa parte delle URL SEO del catalogo.
# L'indicizzazione è governata da: canonical + meta robots (noindex) dove necessario.

# Block non-indexable areas (align to KeepStore real pages)
Disallow: /myaccount
Disallow: /myaccount/
Disallow: /myaccount.aspx
Disallow: /carrello.aspx
Disallow: /ordine.aspx
Disallow: /pagamento.aspx
Disallow: /wishlist.aspx
Disallow: /documenti.aspx
Disallow: /documentidettaglio.aspx
Disallow: /datiutente.aspx
Disallow: /cambiapassword.aspx
Disallow: /pay_your_orders.aspx

# Sensitive / internal folders (defense-in-depth)
Disallow: /App_Data/
Disallow: /App_Code/
Disallow: /bin/

# Sitemaps (preferita: sitemap index; mantenuta compatibilità con la sitemap dinamica)
Sitemap: https://www.taikun.it/sitemap.xml
Sitemap: https://www.taikun.it/sitemap.aspx
